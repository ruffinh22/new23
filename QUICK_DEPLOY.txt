🚀 DÉPLOIEMENT RAPIDE (3 ÉTAPES):

1️⃣  SETUP SSH (UNE FOIS):
    
    # Sur votre machine locale:
    ssh-keygen -t rsa -b 4096 -f ~/.ssh/id_rsa -N ""
    ssh-copy-id -i ~/.ssh/id_rsa.pub erpgmc@10.0.5.18
    
    # Tester:
    ssh erpgmc@10.0.5.18 "echo OK"

2️⃣  CONFIGUREZ LE SERVEUR (UNE FOIS):
    
    ssh erpgmc@10.0.5.18
    cd /srv/sgdra
    cp .env.production .env.production.local
    nano .env.production.local          # Adapter les variables
    ./pre-check.sh                      # Vérifier que tout est OK

3️⃣  DÉPLOYEZ DEPUIS VOTRE MACHINE:
    
    cd /chemin/vers/votre/sgdra/
    ./deploy-to-server.sh deploy
    
    # Voilà! Le code est sur le serveur et Docker démarre

🆘 TROUBLESHOOTING RAPIDE:

PROBLÈME: "Permission denied (publickey)"
SOLUTION: 
  ssh-copy-id -i ~/.ssh/id_rsa.pub erpgmc@10.0.5.18
  # Entrer le mot de passe: toor

PROBLÈME: "rsync command not found on server"
SOLUTION:
  ssh erpgmc@10.0.5.18
  sudo apt-get install rsync

PROBLÈME: "Services ne démarrent pas"
SOLUTION:
  ./deploy-to-server.sh logs backend   # Voir les erreurs
  ./deploy-to-server.sh restart        # Redémarrer

PROBLÈME: ".env.production.local not found"
SOLUTION:
  # Le script crée automatiquement depuis .env.production
  # Vous devez l'éditer sur le serveur:
  ssh erpgmc@10.0.5.18
  nano /srv/sgdra/.env.production.local
